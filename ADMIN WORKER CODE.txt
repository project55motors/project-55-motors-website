export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const method = request.method;

    // --- ROUTE NORMALISATION ---
    let path = url.pathname;
    if (path.startsWith("/api/")) path = path.slice(4);
    if (path === "/api") path = "/";

    // --- ENV VARS ---
    const BASE = env.AIRTABLE_BASE_ID;
    const TABLE = env.AIRTABLE_TABLE_NAME || "Cars";
    const PAT   = env.AIRTABLE_PAT;
    const ADMIN_USER = env.ADMIN_USER;
    const ADMIN_PASS = env.ADMIN_PASS;

    const AIRTABLE_URL = `https://api.airtable.com/v0/${BASE}/${encodeURIComponent(TABLE)}`;

    // --- CORS ---
    const origin = request.headers.get("Origin") || "";
    const allowed = [
      "https://project55motors.co.uk",
      "https://www.project55motors.co.uk"
    ];

    const cors = {
      "Access-Control-Allow-Origin": allowed.includes(origin)
        ? origin
        : "https://project55motors.co.uk",
      "Access-Control-Allow-Methods": "GET,POST,OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type",
      "Access-Control-Allow-Credentials": "true"
    };

    if (method === "OPTIONS")
      return new Response(null, { status: 204, headers: cors });

    // --- COOKIES ---
    const cookies = Object.fromEntries(
      (request.headers.get("Cookie") || "")
        .split(";")
        .map(c => c.trim().split("="))
        .filter(([k]) => k)
    );

    const isLoggedIn = cookies.admin_session === "1";
    const loginCookie  = "admin_session=1; Path=/; Secure; SameSite=None; Max-Age=86400";
    const logoutCookie = "admin_session=0; Path=/; Secure; SameSite=None; Max-Age=0";

    // --------------------------------------------------------------------
    //  PUBLIC ENDPOINT — USED BY cars.js on Home + Stock pages
    // --------------------------------------------------------------------
    if (path === "/cars" && method === "GET") {
      const res = await fetch(AIRTABLE_URL, {
        headers: { Authorization: `Bearer ${PAT}` }
      });

      if (!res.ok) {
        return new Response(JSON.stringify({ error: "Airtable error" }), {
          status: 500,
          headers: cors
        });
      }

      const data = await res.json();

      return new Response(
        JSON.stringify({ records: data.records }),
        { headers: { ...cors, "Content-Type": "application/json" } }
      );
    }

    // --------------------------------------------------------------------
    //  LOGIN
    // --------------------------------------------------------------------
    if (path === "/login" && method === "POST") {
      const { username, password } = await request.json();

      if (username === ADMIN_USER && password === ADMIN_PASS) {
        return new Response(JSON.stringify({ success: true }), {
          headers: { ...cors, "Set-Cookie": loginCookie }
        });
      }

      return new Response(JSON.stringify({ error: "Invalid" }), {
        status: 401,
        headers: cors
      });
    }

    // LOGOUT
    if (path === "/logout" && method === "POST") {
      return new Response(JSON.stringify({ success: true }), {
        headers: { ...cors, "Set-Cookie": logoutCookie }
      });
    }

    // --------------------------------------------------------------------
    //  ADMIN AUTH REQUIRED BELOW
    // --------------------------------------------------------------------
    if (!isLoggedIn) {
      return new Response(JSON.stringify({ error: "Not logged in" }), {
        status: 401,
        headers: cors
      });
    }

    // --------------------------------------------------------------------
    //  GET ALL CARS (ADMIN TABLE)
    // --------------------------------------------------------------------
    if (path === "/admin/all" && method === "GET") {
      const res = await fetch(AIRTABLE_URL, {
        headers: { Authorization: `Bearer ${PAT}` }
      });

      if (!res.ok) {
        return new Response("Airtable error", {
          status: 500,
          headers: cors
        });
      }

      const { records } = await res.json();
      let html = "";

      for (const r of records) {
        const f = r.fields;

        const photo = f.Photos?.[0]?.url || "";
        const img = photo ? `<img src="${photo}" class="photo-thumb">` : "—";

        const statuses = ["Available", "Sold", "Hidden"]
          .map(s => `<option ${f.Status === s ? "selected" : ""}>${s}</option>`)
          .join("");

        html += `
<tr data-id="${r.id}">
  <td>${img}</td>
  <td><input name="Make_Model" value="${f.Make_Model || ""}"></td>
  <td><input name="Registration" value="${f.Registration || ""}"></td>
  <td><input name="Price" type="number" value="${f.Price || 0}"></td>
  <td><input name="Mileage" type="number" value="${f.Mileage || 0}"></td>
  <td><input type="date" name="MOT_Date" value="${f.MOT_Date || ""}"></td>
  <td><select name="Status">${statuses}</select></td>
  <td><textarea name="Short_Description">${f.Short_Description || ""}</textarea></td>
  <td><textarea name="Full_Description">${f.Full_Description || ""}</textarea></td>
  <td>
    <button class="btn btn-save" onclick="save('${r.id}')">Save</button><br>
    <button class="btn btn-sold" onclick="sold('${r.id}')">Sold</button>
  </td>
</tr>`;
      }

      return new Response(html, {
        headers: { ...cors, "Content-Type": "text/html" }
      });
    }

    // --------------------------------------------------------------------
    //  UPDATE CAR
    // --------------------------------------------------------------------
    if (path === "/admin/update" && method === "POST") {
      const body = await request.json();

      const res = await fetch(`${AIRTABLE_URL}/${body.id}`, {
        method: "PATCH",
        headers: {
          "Authorization": `Bearer ${PAT}`,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({ fields: body.fields })
      });

      return res.ok
        ? new Response("OK", { headers: cors })
        : new Response("Fail", { status: 500, headers: cors });
    }

    // --------------------------------------------------------------------
    //  CREATE CAR
    // --------------------------------------------------------------------
    if (path === "/admin/create" && method === "POST") {
      const res = await fetch(AIRTABLE_URL, {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${PAT}`,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          records: [
            { fields: { Make_Model: "New Vehicle", Status: "Available" } }
          ]
        })
      });

      return res.ok
        ? new Response("OK", { headers: cors })
        : new Response("Fail", { status: 500, headers: cors });
    }

    return new Response("Not found", { status: 404, headers: cors });
  }
}
